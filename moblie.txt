/* ============================================
   OCCASIONS — mobile card carousel dots
   ============================================ */
document.addEventListener('DOMContentLoaded', function () {
  document.querySelectorAll('[data-occ-track]').forEach(function (track) {
    var dots = track.parentNode.querySelectorAll('.occ-dot');
    if (!dots.length) return;
    var raf;
    track.addEventListener('scroll', function () {
      cancelAnimationFrame(raf);
      raf = requestAnimationFrame(function () {
        var cards = track.children, best = 0, bestD = Infinity, mid = track.scrollLeft + track.clientWidth / 2;
        for (var i = 0; i < cards.length; i++) {
          var c = cards[i], d = Math.abs(c.offsetLeft + c.offsetWidth / 2 - mid);
          if (d < bestD) { bestD = d; best = i; }
        }
        dots.forEach(function (d, i) { d.classList.toggle('on', i === best); });
      });
    }, { passive: true });
  });
});
